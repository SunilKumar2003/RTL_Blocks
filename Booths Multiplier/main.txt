module booth_multiplier #(
    parameter N = 8
)(
    input  wire                     clk,
    input  wire                     rst,
    input  wire                     start,

    input  wire signed [N-1:0]      multiplicand,
    input  wire signed [N-1:0]      multiplier,

    output reg  signed [2*N-1:0]    product,
    output reg                      done
);

    reg signed [2*N:0] acc;
    reg signed [N-1:0] mcand;
    reg signed [N-1:0] q;
    reg                 q_minus_1;

    integer count;

    reg signed [2*N:0] acc_temp;

    always @(posedge clk) begin

        if (rst) begin
            acc       <= 0;
            mcand     <= 0;
            q         <= 0;
            q_minus_1 <= 0;
            count     <= 0;
            product   <= 0;
            done      <= 0;
        end

        else begin

            done <= 0;

            if (start) begin

                acc       <= 0;
                mcand     <= multiplicand;
                q         <= multiplier;
                q_minus_1 <= 0;
                count     <= 0;

            end

            else if (count < N) begin

                // Temporary accumulator
                acc_temp = acc;

                // Booth encoding
                case ({q[0], q_minus_1})

                    2'b01:
                        acc_temp = acc + mcand;

                    2'b10:
                        acc_temp = acc - mcand;

                    default:
                        acc_temp = acc;

                endcase

                // Arithmetic right shift
                {acc, q, q_minus_1} <=
                    {acc_temp, q, q_minus_1} >>> 1;

                count <= count + 1;

                // Finished
                if (count == N-1) begin
                    product <= {acc_temp[N-1:0], q};
                    done    <= 1;
                end

            end
        end
    end

endmodule
